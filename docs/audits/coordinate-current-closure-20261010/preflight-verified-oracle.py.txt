def need(condition, cause):
    if not condition:
        raise ValueError(cause)


def same(left, right):
    return json.dumps(left, sort_keys=True, allow_nan=False) == json.dumps(right, sort_keys=True, allow_nan=False)


def read(path, maximum=134217728):
    path = Path(path)
    need(not path.is_absolute() and not {'..', '.git'}.intersection(path.parts), 'own relative closure input')
    full = ROOT / path
    need(full.resolve() == full and full.is_file() and full.stat().st_dev == ROOT.stat().st_dev,
         'plain same-volume closure input')
    for parent in full.parents:
        if parent == ROOT:
            break
        need(not (parent / '.git').exists(), 'own closure repository boundary')
    with full.open('rb') as stream:
        raw = stream.read(maximum + 1)
    need(len(raw) <= maximum, 'bounded complete closure input')
    return raw


def load(path):
    return json.loads(read(path))


def pin(raw):
    return {'bytes': len(raw), 'sha256': hashlib.sha256(raw).hexdigest()}


def dimensions(rows):
    raw = list(rows.values())
    return dict(files=len(raw), bytes_total=sum(map(len, raw)), lines_total=sum(r.count(b'\n') for r in raw),
                bytes_each=max(map(len, raw)), lines_each=max(r.count(b'\n') for r in raw),
                line_bytes_each=max(len(line.rstrip(b'\r')) for r in raw for line in r.split(b'\n')))


def git(*args, data=None):
    run = subprocess.run(['git', '--no-replace-objects', *args], input=data, capture_output=True)
    need(run.returncode == 0 and not run.stderr, 'complete immutable closure Git source')
    return run.stdout


def objects(paths):
    stream = git('cat-file', '--batch', data=''.join(REVISION + ':' + p + '\n' for p in paths).encode())
    bodies, offset = {}, 0
    for path in paths:
        end = stream.index(b'\n', offset)
        _, kind, size = stream[offset:end].split()
        need(kind == b'blob', 'original closure blob kind')
        offset, size = end + 1, int(size)
        bodies[path] = stream[offset:offset + size]
        need(stream[offset + size:offset + size + 1] == b'\n', 'complete binary closure framing')
        offset += size + 1
    need(offset == len(stream), 'whole original closure population consumed')
    return bodies


def literal(raw, name):
    found = [ast.literal_eval(node.value) for node in ast.parse(raw).body if isinstance(node, ast.Assign)
             and any(isinstance(target, ast.Name) and target.id == name for target in node.targets)]
    need(len(found) == 1, 'unique original closure guide')
    return found[0]


def swap(text, old, new):
    need(text.count(old) == 1, 'whole exact closure guide seam')
    return text.replace(old, new, 1)


def allocation(canonical):
    answers = {}
    for path, raw in canonical.items():
        text = raw.decode()
        if not text.startswith('---\n'):
            continue
        front = text.split('\n---', 1)[0] + '\n'
        questions = []
        for entry in re.finditer(r'^answers:[ \t]*(.*)\n((?:[ \t]+.*\n)*)', front, re.M):
            questions.extend(json.loads(entry[1]) if entry[1] else [q.strip().removeprefix('"').removesuffix('"')
                for q in re.findall(r'^ +\- +(.*)$', entry[2], re.M)])
        if not questions:
            continue
        identifiers = re.findall(r'^id: +(.+)$', front, re.M)
        need(len(identifiers) == 1, 'unique complete source question identity')
        identifier = identifiers[0].strip().strip('"')
        for question in questions:
            answers.setdefault(question, []).append((identifier, path))
    return ''.join(question + '\t' + ','.join(i for i, _ in sorted(rows)) + '\t'
                   + ','.join(p for _, p in sorted(rows)) + '\n' for question, rows in sorted(answers.items()))


def semantic(actual, expected_stdout):
    need(type(actual['status']) is int and actual['status'] == 0, 'fresh complete semantic integer success')
    need(actual['command'] == ['scripts/run_with_ram_guard.sh', '--', 'python3', '-B', READER]
         and actual['stdout'] == expected_stdout, 'whole fresh original semantic command and result')
    need(re.fullmatch(r'ram-guard: PID [0-9]+; host [0-9.]+%; host cutoff 88%; descendant RSS cutoff 4096 MiB; poll 2s\n',
                     actual['stderr']) is not None, 'whole fresh original semantic guard streams')
    paths = {str(BASE / 'checks.py'), 'scripts/run_with_ram_guard.sh', READER}
    need(actual['instruments'] == {p: pin(read(p) if p.startswith(str(BASE) + '/') else git('show', REVISION + ':' + p))
                                  for p in paths}, 'whole fresh original semantic instruments')


def source_snapshot(snapshot, canonical, chapters, wanted_module, before):
    need(snapshot['role'] == 'Complete dated final coordinate publication; saved current inputs do not establish later currency.',
         'dated closure cannot establish later currency')
    record = snapshot['record']
    old = json.loads(before[str(PUBLICATION / 'current-input.json')])['record']
    need(set(record) == set(old), 'closed whole final publication input')
    need(record['cards'] == old['cards'] and all(canonical[p].decode() == text for p, text in record['cards'].items()),
         'whole unchanged qualified four-card meanings questions and probes')
    need(record['chapter'] == chapters[FEATURE].decode() and record['tooling_chapter'] == chapters[REFERENCE].decode(),
         'whole final current book source inputs')
    need(same(record['source_members'], {p: pin(raw) for p, raw in canonical.items()}), 'whole final canonical source population')
    measured = dimensions({p: raw for p, raw in canonical.items() if Path(p).parent == Path('docs/knowledge')})
    need(all(type(value) is int for value in record['census'].values()) and same(record['census'], measured),
         'whole typed unchanged current physical census')
    need(same(record['surface'], old['surface']) and same(record['source_budget'], old['source_budget']),
         'whole unchanged current card and source authority')
    rows = allocation(canonical)
    need(set(record['queries']) == {'direct', 'cached'}, 'complete current query roles')
    for role, actual in record['queries'].items():
        need(type(actual['status']) is int and actual['status'] == 0, 'actual final query integer success')
        need(actual['command'] == ['knowledge-map/scripts/query_knowledge_map.sh', '--all'] +
             (['--no-cache'] if role == 'direct' else []) and actual['stdout'] == rows and actual['stderr'] == '',
             'whole independently source-derived exact-case current allocation')
    programmes = snapshot['programmes']
    need(set(programmes) == {MODULE, 'scripts/check_record_value_publication.py', 'scripts/book_source_references.py',
                            't/python/test_record_value_current_publication.py', str(BASE / 'capture.py')},
         'whole final programme population')
    need(programmes[MODULE] == pin(wanted_module.encode()), 'whole final maintained coordinate programme')
    for path in ('scripts/check_record_value_publication.py', 'scripts/book_source_references.py',
                 't/python/test_record_value_current_publication.py'):
        need(programmes[path] == pin(before[path]), 'whole preserved current checker factory and regression programme')
    need(programmes[str(BASE / 'capture.py')] == pin(read(BASE / 'capture.py')), 'whole actual final capture programme')


class Visible(HTMLParser):
    def __init__(self):
        super().__init__()
        self.parts = []

    def handle_data(self, text):
        self.parts.append(text)


def native(raw, blocks):
    parsed = Visible()
    parsed.feed(raw.decode())
    visible = ' '.join(' '.join(parsed.parts).split())
    for block in blocks:
        wanted = block.replace('```bash\n', '').replace('```sh\n', '').replace('```\n', '').replace('`', '')
        need(' '.join(wanted.split()) in visible, 'whole final native coordinate scope and retained gaps')


def task_nodes(raw):
    return {match[1]: match[0] for match in re.finditer(r'^- ID: `([^`]+)`\n.*?(?=^- ID: |\Z)', raw.decode(), re.M | re.S)}
