def need(ok,why):
    if not ok:
        raise ValueError(why)


def pin(raw):
    return {'bytes':len(raw),'sha256':hashlib.sha256(raw).hexdigest()}


def stamp(value):
    return pin(json.dumps(value,sort_keys=True).encode())


def fixed(path):
    root=Path.cwd().resolve();full=root/path
    need(full.resolve()==full and full.is_file() and full.stat().st_dev==root.stat().st_dev,'plain own same-volume hostile input')
    for parent in full.parents:
        if parent==root:
            break
        need(not (parent/'.git').exists(),'own hostile input repository boundary')
    result=subprocess.run(['git','--no-replace-objects','show',REVISION+':'+str(path)],capture_output=True)
    need(result.returncode==0 and not result.stderr and result.stdout==full.read_bytes(),'complete fixed hostile input')
    return result.stdout


def representatives(points):
    selected={}
    for point in points:
        selected.setdefault((point['key'][0],point['selection']['recipe_id']),point)
    return list(selected.values())


def borrow_source(hypotheses,matching,point,recipe_id,raw):
    candidate=copy.deepcopy(hypotheses)
    key=tuple(point['key']);number=point['phase']
    refusal=next(r for r in candidate['refusals'] if (r['mode'],r['domain'],r['index'])==key)
    phase=refusal['phases'][number]
    if point['kind'] in ('origin','load'):
        phase[point['kind']]['caller']['recipe_id']=recipe_id
    else:
        record=next(r for mode in raw['inventories'] if mode['mode']==key[0]
                    for r in mode['domains'][key[1]]['nonempty'] if r['index']==key[2])
        frame_index=[i for i,s in enumerate(record['segments']) if s['kind']=='frame'].index(point['segment_index'])
        case=next(r for r in matching['matches'] if (r['mode'],r['domain'],r['index'])==key)
        binding=next(b for b in case['phases'][number]['callers_inner_to_outer'] if b['frame_index']==frame_index)['source_binding']
        index=phase['active_path_inner_to_outer'].index(binding)
        phase['active_path_inner_to_outer'][index]['caller']['recipe_id']=recipe_id
    return candidate


def semantic_controls(raw,models,matched,points):
    mutations=runpy.run_path(str(BASE/'mutations.py'))
    predecessor=runpy.run_path(str(CLASS/'verify.py'))
    derive=runpy.run_path(str(MATCHING/'match.py'))['derive']
    reconstruct=runpy.run_path(str(MATCHING/'verify.py'))['reconstruct']
    nested=predecessor['public']['context']['nested_records'](models['frame_model']['own'])
    candidates={r['recipe_id']:r for r in matched['expression_alternatives']['candidates']}
    used={r['id'] for r in matched['source_phases']['statements']}
    baseline=predecessor['check'](models,models,raw['inventories'],raw['inventories'],nested)
    controls=[]
    for ordinal,point in enumerate(representatives(points)):
        selection=point['selection'];recipe=selection['recipe_id'];filename=selection['reported_filename']
        other=next(r for key,r in sorted(candidates.items()) if key in used and filename not in r['filename_alternatives'])
        bad_file=other['filename_alternatives'][0]
        bad_line=min(line for row in candidates.values() if filename in row['filename_alternatives']
                     for line in row['line_alternatives'].values() if line not in selection['all_line_alternatives'].values())
        for field,value in [('line',bad_line),('path',bad_file),('source',other['recipe_id'])]:
            if field=='source':
                candidate=raw;affected=[]
                source=borrow_source(matched['source_phases'],matched,point,value,raw)
            else:
                candidate,affected=mutations['mutate'](raw,point,field,value,models)
                source=matched['source_phases']
            qualified=predecessor['check'](models,models,candidate['inventories'],candidate['inventories'],nested)
            observer=mutations['intervention_observer'](matched['observer_qualification'],candidate,qualified)
            failures={}
            for name,programme in [('derive',derive),('reconstruct',reconstruct)]:
                try:
                    programme(candidate,source,qualified,matched['expression_alternatives'],observer)
                except ValueError as error:
                    cause=str(error)
                    expected=('declared own caller coordinate: '+(value if field=='source' else recipe)) if name=='derive' else (
                        'independent declared caller line' if field=='line' else 'independent declared caller filename')
                    need(cause==expected,'intended semantic coordinate refusal: '+name+': '+cause)
                    failures[name]=cause
                else:
                    raise ValueError('false-green semantic coordinate: '+name)
            controls.append({'point':point,'field':field,'value':value,'affected':affected,
                'whole_intervention_original':stamp(candidate),'whole_intervention_source':stamp(source),
                'complete_predecessor_result':stamp(qualified),'refusals':failures})
        (CHECKS/'progress.json').write_text(json.dumps({'completed_representatives':ordinal+1,
            'total_representatives':len(representatives(points)),'completed_controls':len(controls),'stage':'semantic-controls'},sort_keys=True)+'\n')
    return {'baseline_predecessor':stamp(baseline),'controls':controls}


def conserve_original_custody():
    admission=json.loads(fixed(A/'coordinate-original-intake-20261009/admission.json'))
    for name,record in admission['custody']['members'].items():
        path=Path(name);need(path.resolve()==Path.cwd().resolve()/path and path.is_file()
                           and path.stat().st_dev==Path('.').stat().st_dev,'whole original plain same-volume custody')
        digest=hashlib.sha256();size=0
        with path.open('rb') as stream:
            while chunk:=stream.read(1024*1024):
                digest.update(chunk);size+=len(chunk)
        need(record['identity']=={'bytes':size,'sha256':digest.hexdigest()},'every original custody byte conserves')
    return stamp(admission['custody'])
