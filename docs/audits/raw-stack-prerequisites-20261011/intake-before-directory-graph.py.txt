RECOVERY='docs/audits/coordinate-public-route-checks-20261008/drafts.tar.gz'


def recovery(path=Path(RECOVERY)):
    """Recover a required ignored payload against its already committed catalog."""
    catalog_raw=git('show',REVISION+':'+str(Path(RECOVERY).with_suffix('').with_suffix('.json')))
    catalog=json.loads(catalog_raw)
    actual=regular(path,'100644').read_bytes()
    need(pin(actual)==catalog['archive'],'complete recovered archive fixed-catalog byte identity')
    with tarfile.open(path) as archive:
        members=archive.getmembers()
        names=[member.name for member in members]
        need(len(names)==len(set(names)) and set(names)==set(catalog['members'])
             and all(member.isfile() for member in members),'complete recovered archive regular member graph')
        wanted={member.name:pin(archive.extractfile(member).read()) for member in members}
    need(wanted==catalog['members'],'complete recovered archive member byte identity')
    return dict(catalog=pin(catalog_raw),archive=pin(actual),members=wanted,
                provenance='Previously ignored payload; exact committed e428 catalog authenticates recovery. Not a Git blob at that checkpoint.')


def git(*arguments):
    actual=subprocess.run(['git','--no-replace-objects',*arguments],capture_output=True)
    need(type(actual.returncode) is int and actual.returncode==0 and actual.stderr==b'','complete fixed raw prerequisite Git protocol')
    return actual.stdout

    raw=json.loads(git('show',REVISION+':'+str(directory/'record-value-original-raw-20261005/binding.json')))

    address=json.loads(git('show',REVISION+':'+str(directory/'address-raw-admission-20261010/admission.json')))
    selected=set(raw['archives'])|set(classes['inputs'])|set(address['custody']['roots'])
    names=git('ls-tree','-d','--name-only',REVISION+':docs/audits').decode().splitlines()
    families=('observer-','coordinate-','record-value-coordinate-','address-')
    selected.update(str(directory/name) for name in names if name.startswith(families))
    selected.update(str(directory/name) for name in ('record-value-original-raw-20261005',
        'record-value-class-frame-join-20261006','record-value-class-frame-publication-20261006'))

def regular(path,mode):
    root=Path.cwd().resolve();full=root/path
    need(not path.is_absolute() and not {'..','.git'}.intersection(path.parts),'own relative raw prerequisite path')
    need(full.resolve()==full and stat.S_ISREG(full.lstat().st_mode) and full.stat().st_dev==root.stat().st_dev,
         'ordinary same-volume raw prerequisite input')
    for parent in full.parents:
        if parent==root:break
        need(not (parent/'.git').exists(),'raw prerequisite repository boundary')
    need(bool(full.stat().st_mode&0o111)==(mode=='100755'),'complete raw prerequisite executable mode')
    return full


def derive():
    selected=roots();raw=git('--literal-pathspecs','ls-tree','-r','-z',REVISION,'--',*selected)
    entries={}
    for record in raw.rstrip(b'\0').split(b'\0'):
        head,name=record.split(b'\t',1);mode,kind,oid=head.decode().split();path=name.decode()
        need(mode in ('100644','100755') and kind=='blob' and path not in entries,'complete regular raw prerequisite member population')
        entries[path]=dict(mode=mode,oid=oid)
    process=subprocess.Popen(['git','--no-replace-objects','cat-file','--batch'],stdin=subprocess.PIPE,stdout=subprocess.PIPE,stderr=subprocess.PIPE)
    ledger={}
    try:
        for name,entry in sorted(entries.items()):
            path=regular(Path(name),entry['mode']);before=path.stat()
            process.stdin.write(entry['oid'].encode()+b'\n');process.stdin.flush()
            oid,kind,size=process.stdout.readline().split()
            need(oid.decode()==entry['oid'] and kind==b'blob' and int(size)==before.st_size,'whole fixed/current raw prerequisite object')
            remaining=int(size);digest=hashlib.sha256()
            with path.open('rb') as stream:
                while remaining:
                    chunk=process.stdout.read(min(1048576,remaining))
                    need(bool(chunk) and stream.read(len(chunk))==chunk,'whole fixed/current raw prerequisite bytes')
                    digest.update(chunk);remaining-=len(chunk)
                need(stream.read(1)==b'' and process.stdout.read(1)==b'\n','complete raw prerequisite object boundary')
            after=path.stat()
            need((before.st_dev,before.st_ino,before.st_size,before.st_mtime_ns)==
                 (after.st_dev,after.st_ino,after.st_size,after.st_mtime_ns),'complete raw prerequisite source epoch conserved')
            ledger[name]=dict(**entry,bytes=int(size),sha256=digest.hexdigest())
        process.stdin.close();error=process.stderr.read();status=process.wait()
        need(type(status) is int and status==0 and error==b'' and process.stdout.read(1)==b'','complete raw prerequisite binary Git stream')
    finally:
        if process.poll() is None:process.kill();process.wait()
    observed=set()
    for selected_path in selected:
        path=Path(selected_path)
        if path.is_file():observed.add(str(path));continue
        for directory,children,files in os.walk(path,followlinks=False):
            base=Path(directory)
            for child in children:
                candidate=base/child
                need(not candidate.is_symlink() and not (candidate/'.git').exists(),'own raw prerequisite directory graph')
            observed.update(str(base/name) for name in files)
    need(observed==set(entries)|{RECOVERY},'complete current and fixed raw prerequisite member graph plus exact owned recovery')
    recovered=recovery()
    return dict(revision=REVISION,roots=selected,members=ledger,recovery={RECOVERY:recovered},
                task=pin(git('show',REVISION+':docs/tasks/SEMANTIC-INTROSPECTION-MCP-FRONTIER.md')),scope=SCOPE)
