def aggregate(rows):
    return dict(files=len(rows),lines_total=sum(raw.count(b'\n') for raw in rows.values()),
                bytes_total=sum(map(len,rows.values())))


def source():
    names=[name for name in git('ls-tree','-r','--name-only',REVISION,'--','docs/book/src').decode().splitlines()
           if Path(name).parent==Path('docs/book/src') and name.endswith('.md')]
    before={name:git('show',REVISION+':'+name) for name in names};after=dict(before)
    text=before[str(CHAPTER)].decode();need(text.count(ANCHOR)==1 and ADDITION not in text,'unique original prerequisite book seam')
    after[str(CHAPTER)]=text.replace(ANCHOR,ANCHOR+ADDITION,1).encode()
    return before,after


def authority(before,after):
    a,b=aggregate(before),aggregate(after)
    return dict(authority_id='RAW-STACK-PREREQUISITES-20261011',owner=OWNER,baseline=a,
                delta={key:b[key]-a[key] for key in a},
                rationale='Document complete unchanged observer/class-frame predecessor and raw prerequisite acceptance, exact required archive recovery and genuine controls; complete raw semantics and original native/private identities remain open.')


def main():
    parser=argparse.ArgumentParser(description=__doc__);parser.add_argument('--publish',action='store_true');parser.add_argument('--native',action='store_true');args=parser.parse_args()
    before,after=source();current={str(path):path.read_bytes() for path in Path('docs/book/src').glob('*.md')}
    if args.publish:
        need(current==before,'sole owned prerequisite guide mutation')
        CHAPTER.write_bytes(after[str(CHAPTER)])
        rows=[json.loads(line) for line in REGISTRY.read_bytes().splitlines()]

        selected['reference_contract']['aggregate_change']=authority(before,after)
        REGISTRY.write_text(''.join(json.dumps(row,sort_keys=True)+'\n' for row in rows))
        save('book-publication.json',dict(revision=REVISION,before={k:pin(v) for k,v in before.items()},
             after={k:pin(v) for k,v in after.items()},authority=authority(before,after),addition=ADDITION))
        print('PASS sole whole prerequisite book addition and exact fresh aggregate authority; ceilings unchanged.')
    else:
        need(current==after,'complete prerequisite book and unchanged source peers')
        if args.native:

            raw=Path('docs/book/book/13k-isf-feature-support-matrix.html').read_bytes();reader=Text();reader.feed(raw.decode())
            rendered=' '.join(line for line in ADDITION.splitlines() if not line.startswith('```'))
            need(' '.join(''.join(reader.parts).split()).count(' '.join(rendered.split()))==1,'whole unique rendered prerequisite guide')
            need(not (BASE/'book-native.html.gz').exists(),'retain complete first prerequisite native guide')
            (BASE/'book-native.html.gz').write_bytes(gzip.compress(raw,mtime=0))
            save('book-native.json',dict(chapter=pin(raw),addition=pin(ADDITION.encode()),scope='Whole native guidance and source custody; visual QA remains open.'))
        print('PASS complete whole prerequisite book/source/native guide; visual QA remains separately owned.')


if __name__=='__main__':main()
