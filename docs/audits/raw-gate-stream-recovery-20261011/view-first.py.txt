"""Replay the unchanged association reader in a complete ordinary committed-source view."""
import argparse
import importlib.util
import os
from pathlib import Path
import shutil
import subprocess

from capture import BASE, canonical, pin, save

spec = importlib.util.spec_from_file_location('recovery', 'scripts/restore_record_value_gate_streams.py')
recovery = importlib.util.module_from_spec(spec)
spec.loader.exec_module(recovery)
REPO = recovery.REPO
VIEW = Path('.artifacts/raw-gate-stream-recovery-20261011/source-view')
ASSOCIATION = Path('docs/audits/raw-association-acceptance-20261011')
REVISION = '6da1d3df3062830b730e550fde799acfab8f2cb4'


def read(path):
    import json
    return json.loads(recovery.local(path).read_bytes())


def file_identity(path):
    import hashlib
    path = recovery.local(path)
    before = path.stat()
    digest = hashlib.sha256()
    with path.open('rb') as stream:
        while chunk := stream.read(1048576):
            digest.update(chunk)
    after = path.stat()
    recovery.require((before.st_ino, before.st_dev, before.st_size, before.st_mtime_ns) ==
                     (after.st_ino, after.st_dev, after.st_size, after.st_mtime_ns), 'complete view file epoch')
    return dict(bytes=before.st_size, sha256=digest.hexdigest(), executable=bool(before.st_mode & 0o111))


def source():
    members = {n: dict(bytes=r['bytes'], sha256=r['sha256'], executable=r['mode'] == '100755')
               for n, r in read(ASSOCIATION / 'intake-before.json')['members'].items()}
    result = subprocess.run(['git', '--no-replace-objects', 'ls-tree', '--full-tree', '-r', '-z',
                             REVISION, '--', str(ASSOCIATION)], capture_output=True)
    recovery.require(result.returncode == 0 and result.stderr == b'', 'complete own fixed source Git protocol')
    for item in result.stdout.rstrip(b'\0').split(b'\0'):
        header, name = item.split(b'\t', 1)
        mode, kind, oid = header.decode().split()
        recovery.require(mode in ('100644', '100755') and kind == 'blob', 'ordinary complete original reader member')
        name = name.decode()
        native = subprocess.run(['git', '--no-replace-objects', 'cat-file', 'blob', oid], capture_output=True)
        recovery.require(native.returncode == 0 and native.stderr == b'', 'whole original reader Git body')
        row = dict(**pin(native.stdout), executable=mode == '100755')
        recovery.require(file_identity(Path(name)) == row, 'whole unchanged committed reader source')
        recovery.require(name not in members, 'disjoint complete original reader namespace')
        members[name] = row
    return dict(revision=REVISION, members=members,
        condition='Whole original source graph and complete unchanged original reader namespace. Ordinary same-volume view; no nested Git repository. Read-only history resolves from the owning repository through explicit runtime-derived Git directory/work-tree settings.')


def census(streams=False):
    declared = read(BASE / 'source.json')['members']
    expected = dict(declared)
    if streams:
        rows = recovery.plan(recovery.committed_receipts([str(ASSOCIATION / 'checks.json')]), str(VIEW))
        recovery.require(all(r['present'] for r in rows), 'all recovered streams physically complete')
        expected.update({str(r['path'].relative_to(VIEW)): dict(**pin(r['body']), executable=False) for r in rows})
    files, directories = {}, {'.'}
    for path in sorted((REPO / VIEW).rglob('*')):
        relative = path.relative_to(REPO)
        recovery.local(relative)
        if path.is_dir():
            directories.add(str(path.relative_to(REPO / VIEW)))
        else:
            files[str(path.relative_to(REPO / VIEW))] = file_identity(relative)
    wanted = {'.'} | {str(p) for n in expected for p in Path(n).parents}
    recovery.require(files == expected and directories == wanted, 'whole view bytes/modes/file/directory population')
    return dict(members=files, directories=sorted(directories))


def materialize():
    recovery.require(not recovery.local(VIEW).exists(), 'fresh complete source view before first materialization')
    sources = source()
    # Preflight every original source and destination before copying any body.
    for name, row in sources['members'].items():
        recovery.require(file_identity(Path(name)) == row, 'whole original source before copy')
        recovery.local(VIEW / name)
    recovery.local(VIEW).mkdir(parents=True)
    for name, row in sorted(sources['members'].items()):
        target = recovery.local(VIEW / name)
        target.parent.mkdir(parents=True, exist_ok=True)
        shutil.copyfile(REPO / name, target)
        target.chmod(0o755 if row['executable'] else 0o644)
    save('source.json', sources)
    population = census()
    save('view-before-streams.json', population)
    print('PASS complete committed-source ordinary view; ignored streams absent.')


def reader(role):
    restored = role == 'after'
    before = census(restored)
    command = ['python3', '-B', str(ASSOCIATION / 'verify.py')]
    env = dict(os.environ, GIT_DIR=str(REPO / '.git'), GIT_WORK_TREE=str(REPO / VIEW))
    prefix = subprocess.run(['git', 'rev-parse', '--show-prefix'], cwd=REPO / VIEW,
                            env=env, capture_output=True)
    recovery.require(prefix.returncode == 0 and prefix.stdout == b'' and prefix.stderr == b'',
                     'explicit ordinary work-tree root for original read-only Git commands')
    programmes = {str(p.relative_to(REPO / VIEW)): dict(body=(raw := p.read_bytes()).decode(), identity=pin(raw))
                  for p in sorted((REPO / VIEW / ASSOCIATION).glob('*.py'))}
    stdout = BASE / ('original-' + role + '.stdout.txt')
    stderr = BASE / ('original-' + role + '.stderr.txt')
    started = dict(command=command, cwd=str(VIEW), git_directory='.git', git_work_tree=str(VIEW),
        programmes=programmes, inputs=before, stdout_path=str(stdout), stderr_path=str(stderr),
        condition='Exact original complete reader, original source and dated role. Native terminal saved before judgment; source/oracle bytes unchanged.')
    save('original-' + role + '-started.json', started)
    with stdout.open('xb', buffering=0) as out, stderr.open('xb', buffering=0) as err:
        result = subprocess.run(command, cwd=REPO / VIEW, env=env, stdout=out, stderr=err)
    actual = dict(**started, status=result.returncode, stdout=stdout.read_text(), stderr=stderr.read_text(),
                  inputs_after=census(restored))
    save('original-' + role + '-actual.json', actual)
    recovery.require(actual['inputs'] == actual['inputs_after'], 'whole original source/recovery epoch across reader')
    if restored:
        recovery.require(type(result.returncode) is int and result.returncode == 0
            and actual['stdout'] == 'PASS unchanged complete original raw-body association acceptance; full raw semantic parent, canonical publication and original native/private scopes remain open.\n'
            and actual['stderr'] == '', 'actual unchanged original complete association reader success after recovery')
    else:
        recovery.require(type(result.returncode) is int and result.returncode == 1 and actual['stdout'] == ''
            and 'line 77, in actual' in actual['stderr']
            and actual['stderr'].endswith('ValueError: complete ordinary admission member\n'),
            'actual intended unchanged original missing-stream refusal')
    print('Original complete association reader native status', result.returncode)
    raise SystemExit(result.returncode)


if __name__ == '__main__':
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--materialize', action='store_true')
    parser.add_argument('--reader', choices=('before', 'after'))
    args = parser.parse_args()
    recovery.require(args.materialize != bool(args.reader), 'one explicit source-view role')
    materialize() if args.materialize else reader(args.reader)
