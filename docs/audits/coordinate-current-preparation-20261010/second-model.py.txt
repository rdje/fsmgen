def parts(raw):
    front, body = raw.decode().rsplit('---\n', 1)
    return front, body.strip().split('\n\n')


def proposal(path, original):
    front, paragraphs = parts(original)
    front = front.replace(next(line for line in front.splitlines() if line.startswith('date:')),
                          'date: 2026-10-10', 1)
    if path == CORE:
        front = front.replace('date: 2026-10-10\n',
                              '  - "' + QUESTION + '"\ndate: 2026-10-10\n', 1)
        front = front.replace('evidence: >-\n', 'evidence: >-\n  docs/audits/coordinate-original-hostile-20261009/scope.txt;\n', 1)
    if path == PARSER:
        if front.count(PARSER_OLD) != 1:
            raise ValueError('one original parser current route')
        front = front.replace(PARSER_OLD, PARSER_SNAPSHOT + ' &&\n  ' + CURRENT, 1)
    paragraphs = [REPLACEMENTS.get(path, {}).get(i, text) for i, text in enumerate(paragraphs)]
    if path in ADDITIONS:
        paragraphs.append(ADDITIONS[path])
    wrapped = [fill(' '.join(p.split()), width=120, break_long_words=False,
                    break_on_hyphens=False) for p in paragraphs]
    return (front + '---\n\n' + '\n\n'.join(wrapped) + '\n').encode()
