def need(ok, cause):
    if not ok:
        raise ValueError(cause)


def same(left, right):
    return json.dumps(left, sort_keys=True, allow_nan=False) == json.dumps(right, sort_keys=True, allow_nan=False)


def pin(raw):
    return {'bytes': len(raw), 'lines': raw.count(b'\n'), 'sha256': hashlib.sha256(raw).hexdigest()}


def git(*args, data=None):
    result = subprocess.run(['git', '--no-replace-objects', *args], input=data, capture_output=True)
    need(result.returncode == 0 and not result.stderr, 'whole independent publication Git source')
    return result.stdout


def objects(paths):
    output = git('cat-file', '--batch', data=''.join(REVISION + ':' + p + '\n' for p in paths).encode())
    result, offset = {}, 0
    for path in paths:
        end = output.index(b'\n', offset)
        _, kind, length = output[offset:end].split()
        need(kind == b'blob', 'complete original publication blob kind')
        offset, length = end + 1, int(length)
        result[path] = output[offset:offset + length]
        need(output[offset + length:offset + length + 1] == b'\n', 'whole original publication framing')
        offset += length + 1
    need(offset == len(output), 'complete independent original publication consumption')
    return result


def metrics(rows):
    raw = list(rows.values())
    return dict(files=len(raw), bytes_total=sum(map(len,raw)), lines_total=sum(r.count(b'\n') for r in raw),
        bytes_each=max(map(len,raw)), lines_each=max(r.count(b'\n') for r in raw),
        line_bytes_each=max(len(line.rstrip(b'\r')) for r in raw for line in r.split(b'\n')))


def counts(rows):
    return {key: metrics(rows)[key] for key in ('files', 'bytes_total', 'lines_total')}


def source_check(captured, canonical, proposal):
    record = captured['record']
    wanted = {p: row['text'] for p,row in proposal['cards'].items()}
    need(record['cards'] == wanted and all(canonical[p].decode() == text for p,text in wanted.items()),
         'whole four-card approved source and meaning epoch')
    expected_pins = {p:{k:pin(raw)[k] for k in ('bytes','sha256')} for p,raw in canonical.items()}
    need(same(record['source_members'], expected_pins), 'whole independent current canonical population')
    physical = {p:r for p,r in canonical.items() if Path(p).parent == Path('docs/knowledge')}
    need(same(record['census'], metrics(physical)), 'whole independently measured current card dimensions')
    target = []
    for path, text in wanted.items():
        front = text.rsplit('---\n',1)[0]
        identifier = re.findall(r'^id: (.+)$', front, re.M)
        need(len(identifier) == 1, 'unique independently derived fact identity')
        for question in re.findall(r'^  - "([^"\n]+)"$', front, re.M):
            target.append(' '.join(question.lower().split()) + '\t' + identifier[0] + '\t' + path)
    ids = {Path(p).stem for p in wanted}
    for role, actual in record['queries'].items():
        need(role in ('direct','cached') and type(actual['status']) is int and actual['status'] == 0,
             'actual distinct typed current query success')
        command = ['knowledge-map/scripts/query_knowledge_map.sh','--all'] + (['--no-cache'] if role == 'direct' else [])
        need(actual['command'] == command and actual['stderr'] == '', 'actual distinct current query command and streams')
        rows = actual['stdout'].splitlines()
        need(all(len(row.split('\t')) == 3 for row in rows), 'complete current query row kinds')
        owned = sorted(row for row in rows if ids.intersection(row.split('\t')[1].split(',')))
        need(owned == sorted(target), 'every independently reconstructed current target question')
    need(set(record['queries']) == {'direct','cached'}
         and record['queries']['direct']['stdout'] == record['queries']['cached']['stdout'],
         'complete distinct direct/cache global query conservation')
    for path, expected in captured['programmes'].items():
        need({k:pin(Path(path).read_bytes())[k] for k in ('bytes','sha256')} == expected,
             'complete actual current programme epoch')


def receipt(actual, command, stdout=None, saved_sources=None):
    need(type(actual['status']) is int and actual['status'] == 0, 'actual publication integer success')
    need(actual['command'] == ['scripts/run_with_ram_guard.sh','--',*command], 'whole actual publication command')
    instruments = {str(BASE/'checks.py'),'scripts/run_with_ram_guard.sh'} | {p for p in command if Path(p).is_file()}
    saved_sources = saved_sources or {}
    need(actual['instruments'] == {p:{k:pin(saved_sources.get(p, Path(p).read_bytes()))[k]
        for k in ('bytes','sha256')} for p in instruments},
         'whole actual publication instruments')
    need(type(actual['stdout']) is str and type(actual['stderr']) is str
         and 'host cutoff 88%; descendant RSS cutoff 4096 MiB; poll 2s' in actual['stderr'],
         'whole actual publication streams and unchanged guard')
    if stdout is not None:
        need(actual['stdout'] == stdout, 'whole actual publication stdout')


class Visible(HTMLParser):
    def __init__(self):
        super().__init__()
        self.parts = []

    def handle_data(self, data):
        self.parts.append(data)


def native(html, blocks):
    parsed = Visible()
    parsed.feed(html)
    visible = ' '.join(' '.join(parsed.parts).split())
    for block in blocks:
        text = block.replace('```bash\n','').replace('```sh\n','').replace('```\n','').replace('`','')
        need(' '.join(text.split()) in visible, 'whole native current scope and remaining gaps')
