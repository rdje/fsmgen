def same(a, b):
    if type(a) is not type(b):
        return False
    if isinstance(a, dict):
        return a.keys() == b.keys() and all(same(a[k], b[k]) for k in a)
    if isinstance(a, list):
        return len(a) == len(b) and all(same(x, y) for x, y in zip(a, b))
    return a == b


def plain(value):
    if isinstance(value, list) and len(value) == 2:
        if value[0] == 'hash':
            return {k: plain(v) for k, v in value[1].items()}
        if value[0] == 'array':
            return [plain(v) for v in value[1]]
    return value


def sexps(text):
    roots, stack = [], []
    for token in re.findall(r'\(|\)|[^\s()]+', text):
        if token == '(':
            node = []
            (stack[-1] if stack else roots).append(node)
            stack.append(node)
        elif token == ')':
            assert stack, 'unbalanced closing parenthesis'
            stack.pop()
        else:
            assert stack, ('token outside expression', token)
            stack[-1].append(token)
    assert not stack, 'unbalanced opening parenthesis'
    return roots


def named(rows, name):
    found = [row for row in rows if isinstance(row, list) and row and row[0] == name]
    assert len(found) == 1, ('expected unique expression', name)
    return found[0]

def assignment(container, lhs, op, rhs, kind=None, guard=None):
    value = {'lhs': lhs, 'op': op, 'rhs': rhs}
    if kind is not None:
        value['source_kind'] = kind
    if guard is not None:
        value['guard'] = {'expr': guard}
    return container, value

def qualify(index, graph, files):
    module = plain(graph)
    expected_name = 'coverage' if index in (0, 1, 2) else 'rule_coverage' if index == 8 else 'monitor_coverage'
    assert module['actor_name'] == expected_name
    scopes = {None: module, **module['children']}
    assert set(module['children']) == ({'worker'} if index in (0, 1, 2) else set())
    field_paths = set()
    carrier = {name: sexps(body)[0] for name, body in files.items()}
    for child, family, (container_name, expected) in specs(index):
        scope = scopes[child]
        prefix = '/1' if child is None else '/1/children/1/' + child + '/1'
        containers = scope[family]
        ci = next(i for i, c in enumerate(containers) if c['name'] == container_name)
        container = containers[ci]
        assert sum(c['name'] == container_name for c in containers) == 1
        matches = [(i, a) for i, a in enumerate(container['assignments'])
                   if a['lhs'] == expected['lhs'] and a.get('source_kind') == expected.get('source_kind')]
        assert len(matches) == 1, (index, container_name, expected['lhs'])
        ai, actual = matches[0]
        assert same(actual, expected), ('numeric producer assignment', index, container_name, expected, actual)
        field_paths.add(f'{prefix}/{family}/1/{ci}/1/assignments/1/{ai}/1/rhs')
        candidates = [(i, p) for i, p in enumerate(scope['assignment_provenance'])
                      if p['activation']['container_name'] == container_name and p['assignment_index'] == ai]
        assert len(candidates) == 1
        pi, provenance = candidates[0]
        assert provenance['target'] == expected['lhs'] and provenance['operator'] == expected['op']
        assert same(provenance['rhs'], expected['rhs']), ('numeric provenance', index, container_name)
        assert provenance['source_kind'] == (expected.get('source_kind') or 'scheduler_can_accept')
        assert provenance['activation']['container_kind'] == ('state' if family == 'states' else 'dt')
        assert same(provenance['activation']['assignment_guard'], expected.get('guard'))
        field_paths.add(f'{prefix}/assignment_provenance/1/{pi}/1/rhs')
        file = (expected_name if child is None else child) + '.fsm'
        assert carrier[file][0] == '?fsm:' + (expected_name if child is None else child)
        block = named(carrier[file], container_name if family == 'states' else '-' + container_name)
        lhs = expected['lhs'] + ('>' if any(p['name'] == expected['lhs'] and p['direction'] == 'output' for p in scope['ports']) else '')
        emitted = [expected['op'], [lhs, str(expected['rhs'])]]
        if 'guard' in expected:
            emitted += ['<', sexps(expected['guard']['expr'])[0]]
        assert sum(same(node, emitted) for node in block[1:]) == 1, ('carrier assignment', index, container_name, emitted)
    # Widths are independently sourced from interface declarations, not counts.
    widths = ['data', 'result'] if index in (0, 1, 2) else ['din', 'observed'] if index == 8 else ['observed']
    source_actor = authored[0 if index in (0, 1, 2) else 1 if index == 8 else 2]
    interface = named(source_actor, 'interface')
    for child, scope in scopes.items():
        prefix = '/1' if child is None else '/1/children/1/' + child + '/1'
        file = (expected_name if child is None else child) + '.fsm'
        sizes = named(carrier[file], '+size')
        for name in widths:
            declaration = next(p for p in interface[1:] if p[1] == name)
            assert declaration[2:] == [['width', '4']]
            assert same(scope['signal_widths'][name], 4), ('declared width kind/value', index, child, name)
            if child is None:
                port = next(p for p in scope['ports'] if p['name'] == name)
                assert port['direction'] == declaration[0] and port['width'] == '4'
                assert named(sizes, name) == [name, '4']
            else:
                # The complete actor width lookup is inherited by the child;
                # its public ports and emitted declarations are use-selected.
                assert not any(p['name'] == name for p in scope['ports'])
                assert not any(isinstance(p, list) and p[0] == name for p in sizes[1:])
                interface_node = named(carrier[file], '+interface')
                assert not any(isinstance(p, list) and p[1] == name for p in interface_node[1:])
            field_paths.add(prefix + '/signal_widths/1/' + name)
    assert field_paths == {row[0] for row in roster[index]['differences']}, ('complete field contract', index)
    return field_paths
