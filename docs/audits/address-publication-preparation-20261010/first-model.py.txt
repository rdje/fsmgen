def parts(raw):
    front, body = raw.decode().rsplit('---\n', 1)
    return front, body.strip().split('\n\n')


def proposal(path, original):
    front, paragraphs = parts(original)
    if path == CORE:
        marker = 'date: 2026-10-10\n'
        if front.count(marker) != 1:
            raise ValueError('unique original address fact date')
        front = front.replace(marker, '  - "' + QUESTION + '"\n' + marker, 1)
        front = front.replace('evidence: >-\n', 'evidence: >-\n  docs/audits/address-parent-acceptance-20261010/scope.txt;\n', 1)
    proposed = [REPLACEMENTS.get(path, {}).get(i, p) for i, p in enumerate(paragraphs)]
    if path == CORE:
        proposed.append(ADDITION)
    wrapped = [fill(' '.join(p.split()), width=120, break_long_words=False, break_on_hyphens=False) for p in proposed]
    # Joining Markdown links preserves their intended original endpoints.
    text = front + '---\n\n' + '\n\n'.join(wrapped) + '\n'
    text = text.replace('[Observer/helper source meanings] (', '[Observer/helper source meanings](')
    text = text.replace('[source admission] (', '[source admission](')
    return text.encode()
